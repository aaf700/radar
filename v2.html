    <script>

        let userLat = 25.2048; 

        let userLon = 55.2708;

        

        const RADIUS_NM = 22; 

        const MAX_DISTANCE_KM = 40.0;

        

        const REFRESH_RATE = 5000; 

        const CYCLE_RATE = 8000; // 8-second hands-free cycle

        

        let map;

        let aircraftMarkers = {};

        let selectedHex = null;

        let currentCycleIndex = 0; 

        

        const routeCache = {}; // Cache to prevent spamming the routing API


        setInterval(() => {

            const now = new Date();

            document.getElementById('clock').innerText = now.toLocaleTimeString('en-US', { hour12: false, hour: '2-digit', minute:'2-digit', second:'2-digit' });

        }, 1000);


        window.onload = () => {

            const loadingText = document.getElementById('loadingText');

            

            if (navigator.geolocation) {

                navigator.geolocation.getCurrentPosition(

                    (pos) => {

                        userLat = pos.coords.latitude;

                        userLon = pos.coords.longitude;

                        loadingText.innerText = "GPS LOCK ACQUIRED...";

                        setTimeout(initializeRadar, 800);

                    },

                    (err) => {

                        loadingText.innerText = "GPS DENIED. USING FALLBACK LOC...";

                        setTimeout(initializeRadar, 1500);

                    },

                    { timeout: 8000 } 

                );

            } else {

                initializeRadar();

            }

        };


        function initializeRadar() {

            document.getElementById('loading').style.opacity = '0';

            setTimeout(() => document.getElementById('loading').style.display = 'none', 500);


            map = L.map('map', {zoomControl: false, attributionControl: false, scrollWheelZoom: false, dragging: false}).setView([userLat, userLon], 10);

            L.tileLayer('https://{s}.tile.openstreetmap.org/{z}/{x}/{y}.png').addTo(map);

            

            L.circleMarker([userLat, userLon], { radius: 6, color: '#00e5ff', fillColor: '#00e5ff', fillOpacity: 1, weight: 2 }).addTo(map);

            L.circle([userLat, userLon], {radius: MAX_DISTANCE_KM * 1000, color: '#00e5ff', weight: 1, fillOpacity: 0.05, dashArray: "5, 10"}).addTo(map);


            fetchAircraft();

            setInterval(fetchAircraft, REFRESH_RATE);

            

            setInterval(() => {

                const hexKeys = Object.keys(aircraftMarkers);

                if(hexKeys.length > 0) {

                    currentCycleIndex = (currentCycleIndex + 1) % hexKeys.length;

                    const nextHex = hexKeys[currentCycleIndex];

                    selectAircraft(aircraftMarkers[nextHex].data);

                }

            }, CYCLE_RATE);

        }


        async function fetchAircraft() {

            const rawUrl = `https://api.adsb.lol/v2/lat/${userLat}/lon/${userLon}/dist/${RADIUS_NM}`;

            

            const urlsToTry = [

                rawUrl, 

                `https://api.allorigins.win/raw?url=${encodeURIComponent(rawUrl)}`,

                `https://api.codetabs.com/v1/proxy?quest=${encodeURIComponent(rawUrl)}`

            ];


            let success = false;

            let targetCount = 0;


            for (const url of urlsToTry) {

                try {

                    const res = await fetch(url);

                    if (!res.ok) continue; 

                    

                    const data = await res.json();

                    if (data.ac) {

                        targetCount = updateMarkers(data.ac);

                        document.getElementById('radarStatus').innerText = `AIRBORNE TARGETS: ${targetCount}`;

                        document.getElementById('radarStatus').style.color = '#00e5ff';

                        success = true;

                        break; 

                    }

                } catch (e) { }

            }


            if (!success) {

                document.getElementById('radarStatus').innerText = `NO SIGNAL`;

                document.getElementById('radarStatus').style.color = '#ffaa00';

            }

        }


        async function fetchFlightRoute(callsign) {

            if (!callsign || callsign.trim() === '') return { from: 'N/A', fromCountry: 'CLASSIFIED', to: 'N/A', toCountry: 'CLASSIFIED' };

            const id = callsign.trim();

            if (routeCache[id]) return routeCache[id];


            try {

                const url = `https://api.flightradar24.com/common/v1/flight/list.json?query=${id}&fetchBy=flight&page=1&limit=1`;

                const proxyUrl = `https://api.allorigins.win/raw?url=${encodeURIComponent(url)}`;

                const res = await fetch(proxyUrl);

                const data = await res.json();

                

                if (data.result && data.result.response && data.result.response.data && data.result.response.data.length > 0) {

                    const flight = data.result.response.data[0];

                    const origin = flight.airport.origin;

                    const dest = flight.airport.destination;

                    

                    const route = {

                        from: origin ? origin.code.iata : 'UNK',

                        fromCountry: origin ? origin.position.country.name : 'Unknown',

                        to: dest ? dest.code.iata : 'UNK',

                        toCountry: dest ? dest.position.country.name : 'Unknown'

                    };

                    routeCache[id] = route;

                    return route;

                }

            } catch(e) { }

            

            const fallback = { from: 'UNK', fromCountry: 'UNAVAILABLE', to: 'UNK', toCountry: 'UNAVAILABLE' };

            routeCache[id] = fallback;

            return fallback;

        }


        function renderIconHtml(ac) {

            const isMil = ac.mil || (ac.dbFlags & 1) || (ac.desc && ac.desc.toLowerCase().includes('mil'));

            const baseColor = isMil ? '#ff3333' : '#00e5ff';

            const track = ac.track || ac.trk || 0;

            const isSelected = selectedHex === ac.hex;

            

            const color = isSelected ? '#ffffff' : baseColor;

            const size = isSelected ? 48 : 36;

            const dropShadow = isSelected ? `drop-shadow(0px 0px 10px rgba(255,255,255,0.8))` : `drop-shadow(0px 0px 6px ${baseColor}80)`;

            const ring = isSelected ? `<div class="selection-ring"></div>` : '';


            return `<div style="position: relative; width: ${size}px; height: ${size}px; display: flex; justify-content: center; align-items: center;">

                ${ring}

                <div style="transform: rotate(${track}deg); transform-origin: center; filter: ${dropShadow}; display: flex; transition: all 0.3s;">

                    <svg viewBox="0 0 24 24" width="${size}" height="${size}" xmlns="http://www.w3.org/2000/svg">

                        <path d="M21 16v-2l-8-5V3.5c0-.83-.67-1.5-1.5-1.5S10 2.67 10 3.5V9l-8 5v2l8-2.5V19l-2 1.5V22l3.5-1 3.5 1v-1.5L13 19v-5.5l8 2.5z" fill="${color}"/>

                    </svg>

                </div>

            </div>`;

        }


        function updateAllIcons() {

            Object.values(aircraftMarkers).forEach(marker => {

                const ac = marker.data;

                const size = selectedHex === ac.hex ? 48 : 36;

                marker.setIcon(L.divIcon({

                    html: renderIconHtml(ac), 

                    className: 'plane-icon', 

                    iconSize: [size, size], 

                    iconAnchor: [size/2, size/2]

                }));

                marker.setZIndexOffset(selectedHex === ac.hex ? 1000 : 0);

            });

        }


        function updateMarkers(aircrafts) {

            const currentHexes = new Set();

            let validCount = 0;

            

            aircrafts.forEach(ac => {

                if(!ac.lat || !ac.lon) return;

                

                // IGNORE AIRCRAFT ON GROUND (Ground status or less than 100 ft)

                if(ac.alt_baro === 'ground' || ac.alt_baro < 100) return;

                

                // 40km Distance Filter

                const distKm = calculateDistance(userLat, userLon, ac.lat, ac.lon);

                if (distKm > MAX_DISTANCE_KM) return;


                currentHexes.add(ac.hex);

                validCount++;


                if(aircraftMarkers[ac.hex]) {

                    aircraftMarkers[ac.hex].setLatLng([ac.lat, ac.lon]);

                    aircraftMarkers[ac.hex].data = ac;

                    

                    // IF THIS IS THE ACTIVE PLANE, TRACK IT LIVE

                    if(selectedHex === ac.hex) {

                        updatePanelData(ac);

                        map.panTo([ac.lat, ac.lon], {animate: true, duration: 1.5}); // Live smooth camera follow

                    }

                } else {

                    const marker = L.marker([ac.lat, ac.lon]).addTo(map);

                    marker.data = ac;

                    aircraftMarkers[ac.hex] = marker;

                }

            });


            // Remove out-of-bounds or landed aircraft

            Object.keys(aircraftMarkers).forEach(hex => {

                if(!currentHexes.has(hex)) {

                    map.removeLayer(aircraftMarkers[hex]);

                    delete aircraftMarkers[hex];

                    if (selectedHex === hex) document.getElementById('infoPanel').classList.remove('active');

                }

            });


            updateAllIcons();

            return validCount;

        }


        async function selectAircraft(ac) {

            selectedHex = ac.hex;

            

            updateAllIcons();

            map.panTo([ac.lat, ac.lon], {animate: true, duration: 1.5});

            

            updatePanelData(ac);

            document.getElementById('infoPanel').classList.add('active');


            // Reset Routing UI State for lookup

            document.getElementById('acFromCode').innerText = '...';

            document.getElementById('acFromCountry').innerText = 'SEARCHING...';

            document.getElementById('acToCode').innerText = '...';

            document.getElementById('acToCountry').innerText = 'SEARCHING...';


            fetchFlightRoute(ac.flight).then(route => {

                // Ensure we haven't cycled to a new plane while fetching

                if (selectedHex === ac.hex) {

                    document.getElementById('acFromCode').innerText = route.from;

                    document.getElementById('acFromCountry').innerText = route.fromCountry;

                    document.getElementById('acToCode').innerText = route.to;

                    document.getElementById('acToCountry').innerText = route.toCountry;

                }

            });


            const imgEl = document.getElementById('acImage');

            imgEl.style.opacity = '0.3';

            

            try {

                const imgUrl = `https://api.planespotters.net/pub/photos/hex/${ac.hex}`;

                let res = await fetch(imgUrl);

                

                if (!res.ok) {

        function calculateDistance(lat1, lon1, lat2, lon2) {

            const R = 6371; 

            const dLat = (lat2 - lat1) * Math.PI / 180;

            const dLon = (lon2 - lon1) * Math.PI / 180;

            const a = Math.sin(dLat/2) * Math.sin(dLat/2) +

                      Math.cos(lat1 * Math.PI / 180) * Math.cos(lat2 * Math.PI / 180) *

                      Math.sin(dLon/2) * Math.sin(dLon/2);

            return R * (2 * Math.atan2(Math.sqrt(a), Math.sqrt(1-a)));

        }
